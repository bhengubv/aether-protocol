   - Little-endian integers (to_le_bytes())

USAGE EXAMPLES
==============

1. Generate identity keys:
   let (priv_key, pub_key) = Ed25519SigningService::generate_keypair();

2. Establish Signal session:
   let mut alice = SignalProtocolService::new();
   let bob_bundle = bob.generate_pre_key_bundle("bob")?;
   alice.process_pre_key_bundle(&bob_bundle)?;

3. Encrypt message:
   let encrypted = alice.encrypt("bob", b"Hello")?;

4. Sign packet:
   let mut packet = MeshPacket::new(PacketType::Data, "alice".to_string());
   signer.sign_packet(&mut packet, &private_key)?;

5. Serialize packet:
   let bytes = PacketSerializer::serialize(&packet)?;

6. Send via transport:
   transport.send_async("peer-uhid", data).await?;
